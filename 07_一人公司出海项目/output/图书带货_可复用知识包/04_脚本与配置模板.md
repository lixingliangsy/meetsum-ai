# 脚本与配置模板（可复用知识包核心）

> 本文件是知识包的"引擎"：`render_topic.py` 是通用竖屏渲染器，`topic_*.json` 是每本书/每平台的配置。在另一台电脑放好这俩 + 6 张背景图 + 书封，就能渲染出片。
> 环境坑见 03；避坑见 05；方法论见 02。

## 一、交付物标准目录结构（每次任务统一）

```
output/<项目代号>/
├── publish/                              ← 最终交付
│   ├── <平台>_a_main_<A时长>s.mp4         ← 主号标准版
│   ├── <平台>_b_sub_<B时长>s.mp4          ← 副号差异版
│   ├── 标题简介总表.md                    ← 12 视频标题/简介/话题/路径
│   └── 发布清单_纯文本.md                 ← 复制粘贴用三段式
├── docs/                                 ← 策划文档
│   └── 全流程prompt_图书带货视频复用指南.md
├── source/                               ← 工程源码与配置
│   ├── ling_planner_pack_<代号>_v1.json  ← 灵枢策划包
│   ├── topic_<代号>_<平台>_<a|b>.json     ← 12 个渲染配置
│   ├── simhei.ttf                        ← 中文字体（或指向系统）
│   └── scripts/
│       ├── render_topic.py               ← 通用竖屏渲染器（本文件下方）
│       ├── gen_<代号>_topics.py          ← topic JSON 生成器
│       ├── render_all_<代号>.py          ← 批量渲染脚本
│       ├── book_cover.png                ← 图书封面（竖版）
│       └── images_b/                     ← B 版差异化背景图
├── images/                               ← A 版通用背景图（6 张）
└── book_cover.png                        ← 图书封面副本
```

**关键约定**：
- 渲染器在 `source/scripts/` 下运行，`BG_DIR` 相对该目录解析（见第三节）。
- A 版配置 `BG_DIR: "../../images"` → 指向根 `images/`；B 版 `BG_DIR: "images_b"` → 指向 `source/scripts/images_b/`。
- 背景图按场景命名 `hook.png/pain.png/method.png/case.png/trust.png/cta.png`，渲染器优先用同名图，否则按池顺序循环。

## 二、topic JSON 配置 Schema + 完整样例

```json
{
  "ORDER": ["hook", "pain", "method", "case", "trust", "cta"],
  "SCENES": {
    "hook":   ["投1元赚68元！400亿操盘手把AI变现拆成8步。"],
    "cta":    ["左下角小黄车，直接带走这本书。", "跟着8步法，跑通你的变现闭环。"],
    "pain":   ["很多人买了一堆AI工具，却变不了现。", "缺的不是工具，是体系。"],
    "method": ["书里8步法：文案、短视频、投放、直播、社群到IP矩阵管理。", "手把手用AI数字员工帮你获客。"],
    "case":   ["投1元赚68元的投放模型，AI跑通变现闭环。"],
    "trust":  ["阿里铁军出身，清华北大特聘AI导师，学员破万。"]
  },
  "BG": {
    "hook":   ["投1元赚68元", "400亿操盘·8步法"],
    "cta":    ["左下角小黄车", "下单跑通闭环"],
    "pain":   ["买了一堆AI工具", "却变不了现"],
    "method": ["8步法闭环", "AI数字员工获客"],
    "case":   ["投1元赚68元", "AI跑通变现"],
    "trust":  ["阿里铁军", "清华北大特聘·学员破万"]
  },
  "BG_DIR": "../../images",
  "OUT": "opc_aishicao_douyin_a.mp4",
  "TAG": "aishicao_douyin_a"
}
```

字段说明：
- `ORDER`：分镜顺序（6 段标准骨架）。
- `SCENES`：每段旁白（数组，支持多句）。
- `BG`：每段叠加在背景上的大字文案（数组，支持多行）。
- `BG_DIR`：背景图目录（相对 `source/scripts/`）。
- `OUT`：输出文件名（建议与最终 `publish/` 名一致，渲染后 mv 过去）。
- `TAG`：中间文件前缀，避免多话题并行互相覆盖。

**B 版差异**：`BG_DIR` 改 `"images_b"`，`SCENES`/`BG` 的 cta/hook 文案换成差异版本，时长靠增大 `SIYU_PAUSE` 拉长。

## 三、render_topic.py（完整代码，直接拷贝即用）

```python
# -*- coding: utf-8 -*-
"""通用话题渲染器（竖屏 1080x1920）。
由 TOPIC_CFG 指向的 JSON 配置驱动：ORDER / SCENES / BG / BG_DIR / OUT / TAG。
复用已验证修复：edge-tts -ar 44100 对齐静音、ASS PlayResX/Y 防 libass 放大、
tpad 不打断音频、eq=brightness=0.15 提亮 + drawbox black@0.10 轻遮罩、大字 borderw=5 + box@0.55。
中间文件统一加 TAG 前缀，支持多话题并行不互相覆盖。
"""
import subprocess, json, os, time, glob

WORK = os.path.dirname(os.path.abspath(__file__))
PY = r"C:\Users\lixingliang\AppData\Local\hermes\hermes-agent\venv\Scripts\python.exe"
FONT = "C\:/Windows/Fonts/simhei.ttf"
W, H, FPS = 1080, 1920, 30
VOICE = "zh-CN-YunxiNeural"
RATE = "+18%"
# 变体支持：SIYU_VARIANT=a|b（默认 a）；SIYU_PAUSE 覆盖句间静音时长（B 版加大以达 60s 目标）
VARIANT = os.environ.get("SIYU_VARIANT", "a")
try:
    PAUSE = float(os.environ.get("SIYU_PAUSE", "0.12"))
except ValueError:
    PAUSE = 0.12
print(f"[variant] SIYU_VARIANT={VARIANT} PAUSE={PAUSE}")

CFG = json.load(open(os.environ["TOPIC_CFG"], encoding="utf-8"))
ORDER = CFG["ORDER"]
SCENES = CFG["SCENES"]
BG = CFG.get("BG", {})
BG_DIR = os.path.join(WORK, CFG.get("BG_DIR", "images"))
OUT_NAME = os.environ.get("SIYU_OUT") or CFG.get("OUT", "opc_topic.mp4")
TAG = CFG.get("TAG", "topic")

# 平台 CTA 覆盖：设置 SIYU_CTA 时，替换 cta 场景的旁白与大字（分号分隔多行）
CTA = os.environ.get("SIYU_CTA", "").strip()
if CTA:
    parts = [p.strip() for p in CTA.split(";") if p.strip()]
    if parts:
        SCENES["cta"] = parts
        BG["cta"] = parts
        print(f"SIYU_CTA override cta scene -> {parts}")

# 背景池
pool = sorted(
    glob.glob(os.path.join(BG_DIR, "*.png")) +
    glob.glob(os.path.join(BG_DIR, "*.jpg")) +
    glob.glob(os.path.join(BG_DIR, "*.jpeg"))
)
def bg_for(idx):
    specific = os.path.join(BG_DIR, f"{ORDER[idx]}.png")
    if os.path.exists(specific):
        return specific
    return pool[idx % len(pool)] if pool else None

def fmt(s):
    h = int(s // 3600); m = int((s % 3600) // 60); sec = s % 60
    return f"{h}:{m:02d}:{sec:05.2f}"

def gen_one(text, out_mp3, retries=6):
    last = None
    for attempt in range(retries):
        try:
            cmd = f'"{PY}" -m edge_tts --voice {VOICE} --rate {RATE} --text "{text}" --write-media "{out_mp3}"'
            subprocess.run(cmd, shell=True, check=True, stderr=subprocess.DEVNULL)
            if os.path.getsize(out_mp3) > 100:
                return float(subprocess.check_output(
                    f'ffprobe -v error -show_entries format=duration -of csv=p=0 "{out_mp3}"').strip())
        except Exception as e:
            last = e
        time.sleep(3 * (attempt + 1))
    raise RuntimeError(f"edge-tts failed after {retries} retries: {text} ({last})")

# ---------- 1) 生成配音 + cues/durs ----------
CUES_PATH = os.path.join(WORK, f"{TAG}_scene_cues.json")
DURS_PATH = os.path.join(WORK, f"{TAG}_scene_durs.json")
VOICE_PATH = os.path.join(WORK, f"{TAG}_voice_full.mp3")

if os.environ.get("SKIP_TTS") == "1" and os.path.exists(CUES_PATH) and os.path.exists(DURS_PATH) and os.path.exists(VOICE_PATH):
    cues = json.load(open(CUES_PATH, encoding="utf-8"))
    durs = json.load(open(DURS_PATH, encoding="utf-8"))
    print("SKIP_TTS=1: reuse existing voice + cues/durs")
else:
    cues, durs, segments = {}, {}, []
    for sc in ORDER:
        cues[sc] = []; t = 0.0
        for i, s in enumerate(SCENES[sc]):
            mp3 = os.path.join(WORK, f"{TAG}_voice_{sc}_{i}.mp3")
            d = gen_one(s, mp3)
            time.sleep(0.8)
            cues[sc].append({"b": round(t, 3), "t": s})
            segments.append((sc, s, mp3, d))
            t += d + PAUSE
        durs[sc] = round(t, 3)

    json.dump(cues, open(CUES_PATH, "w"), ensure_ascii=False, indent=2)
    json.dump(durs, open(DURS_PATH, "w"), ensure_ascii=False, indent=2)

    with open(os.path.join(WORK, f"{TAG}_filelist.txt"), "w") as f:
        for idx, (sc, s, mp3, d) in enumerate(segments):
            wav = os.path.join(WORK, f"{TAG}_v_{idx}.wav")
            subprocess.run(f'ffmpeg -y -i "{mp3}" -ar 44100 "{wav}"', shell=True, check=True, stderr=subprocess.DEVNULL)
            f.write(f"file '{wav}'\n")
            if idx < len(segments) - 1:
                sil = os.path.join(WORK, f"{TAG}_sil_{idx}.wav")
                subprocess.run(f'ffmpeg -y -f lavfi -i anullsrc=r=44100:cl=mono -t {PAUSE} "{sil}"', shell=True, check=True, stderr=subprocess.DEVNULL)
                f.write(f"file '{sil}'\n")

    subprocess.run(
        f'ffmpeg -y -f concat -safe 0 -i "{os.path.join(WORK, TAG + "_filelist.txt")}" -c:a libmp3lame "{VOICE_PATH}"',
        shell=True, check=True, stderr=subprocess.DEVNULL)
    print("voice done; durs=", json.dumps(durs, ensure_ascii=False))

# ---------- 2) 时间轴 ----------
global_off, acc = {}, 0.0
for sc in ORDER:
    global_off[sc] = acc; acc += durs[sc]
TOTAL = acc

# ---------- 3) ASS 字幕 ----------
def wrap_cn(text, n=15):
    if len(text) <= n:
        return text
    return "\\N".join(text[i:i + n] for i in range(0, len(text), n))

ass = """[Script Info]
WrapStyle=2
ScaledBorderAndShadow=yes
ScriptType: v4.00
PlayResX: 1080
PlayResY: 1920

[V4+ Styles]
Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding
Style: C,SimHei,52,&H00FFFFFF,&H000000FF,&H00000000,&H80000000,1,0,0,0,100,100,0,0,4,3,1,2,90,90,120,1
Style: H,SimHei,64,&H00FFFFFF,&H000000FF,&H00000000,&H80000000,1,0,0,0,100,100,0,0,1,4,1,5,90,90,260,1
Style: AIS,SimHei,30,&H00A0A0A0,&H000000FF,&H00000000,&H60000000,0,0,0,0,100,100,0,0,1,2,1,3,50,50,50,1

[Events]
Format: Layer, Start, End, Style, Text
"""
ass += f"Dialogue: 0,0:00:00.00,{fmt(min(5.0, TOTAL))},AIS,本视频含 AI 生成内容\n"
for sc in ORDER:
    sents = cues[sc]; off = global_off[sc]
    for j, it in enumerate(sents):
        start = off + it["b"]
        end = off + sents[j + 1]["b"] if j < len(sents) - 1 else start + max(1.6, len(it["t"]) * 0.18 + 0.5)
        ass += f"Dialogue: 0,{fmt(start)},{fmt(end)},C,{wrap_cn(it['t'])}\n"
open(os.path.join(WORK, f"{TAG}_scene.ass"), "w", encoding="utf-8").write(ass)
print(f"{TAG}_scene.ass written")

# ---------- 4) 逐镜背景视频（cta/method 段叠加真实书封截图） ----------
BOOK_COVER = os.path.join(WORK, "book_cover.png")
HAS_COVER = os.path.exists(BOOK_COVER)
if not HAS_COVER:
    print("book_cover.png NOT found -> skip cover overlay (no error)")

def build_drawtext(lines, y0=None):
    """返回 drawtext 滤镜串（以逗号开头）。y0 为第一行大字基线 y 坐标。"""
    if y0 is None:
        y0 = int(H * 0.32)
    s = ""
    for k, line in enumerate(lines):
        y = y0 + k * 120
        s += (f",drawtext=fontfile='{FONT}':text='{line}'"
               f":fontcolor=white:fontsize=72:x=(w-tw)/2:y={y}"
               f":borderw=5:bordercolor=black@0.85"
               f":shadowcolor=black:shadowx=6:shadowy=6"
               f":box=1:boxcolor=black@0.55:boxborderw=16")
    return s

def bg_input(img):
    if not img:
        return f'-f lavfi -i "color=c=0x6e1010:s={W}x{H}:r={FPS}"'
    return f'-loop 1 -framerate {FPS} -i "{img}"'

def bg_base_chain(label="bg"):
    """背景图基础处理链（缩放+提亮+轻遮罩），输出 [label]。"""
    return (f"[{label}]format=yuv420p,"
            f"scale={W}:{H}:force_original_aspect_ratio=increase:flags=lanczos,crop={W}:{H},"
            f"eq=brightness=0.15:contrast=1.06,"
            f"drawbox=y=0:color=black@0.10:w={W}:h={H}:t=fill[bg]")

concat = []
for i, sc in enumerate(ORDER):
    lines = BG.get(sc, [])
    dur = durs[sc]
    img = bg_for(i)
    out = os.path.join(WORK, f"{TAG}_bg_{sc}.mp4")

    # ---- CTA：真实书封居中清晰叠加（白描边 + 轻阴影），CTA 大字置于书封下方保持可读 ----
    if sc == "cta" and HAS_COVER:
        input_src = bg_input(img) + f' -loop 1 -framerate {FPS} -i "{BOOK_COVER}"'
        cover_chain = (
            f"[1:v]format=yuva420p,scale=-1:700:flags=lanczos[s0];"
            f"[s0]pad=iw+40:ih+40:(ow-iw)/2:(oh-ih)/2:color=white[cvw];"
            f"[s0]gblur=sigma=16,colorchannelmixer=rr=0:gg=0:bb=0:aa=0.5[sh0];"
            f"[sh0]pad=iw+40:ih+40:(ow-iw)/2:(oh-ih)/2:color=0x00000000[sh1]"
        )
        overlays = (
            f"[bg][sh1]overlay=(W-w)/2+18:222[bg1];"
            f"[bg1][cvw]overlay=(W-w)/2:200[base]"
        )
        dt = build_drawtext(lines, y0=1040)
        chain = dt[1:] if dt.startswith(",") else dt
        fc = bg_base_chain("0:v") + ";" + cover_chain + ";" + overlays + ";" + f"[base]{chain}[out]"
        cmd = (f'ffmpeg -y {input_src} -filter_complex "{fc}" '
               f'-map "[out]" -t {dur} -r {FPS} -pix_fmt yuv420p -movflags +faststart "{out}"')
        print("render bg (cta+cover):", sc, dur)
        subprocess.run(cmd, shell=True, check=True, stderr=subprocess.DEVNULL)
        concat.append(f"file '{TAG}_bg_{sc}.mp4'")
        continue

    # ---- METHOD：右下角小号书封（高 420）强化「这本书」存在感 ----
    if sc == "method" and HAS_COVER:
        input_src = bg_input(img) + f' -loop 1 -framerate {FPS} -i "{BOOK_COVER}"'
        cover_chain = (
            f"[1:v]format=yuva420p,scale=-1:420:flags=lanczos[s0];"
            f"[s0]pad=iw+28:ih+28:(ow-iw)/2:(oh-ih)/2:color=white[cvw]"
        )
        overlays = f"[bg][cvw]overlay=W-w-70:H-h-70[base]"
        dt = build_drawtext(lines)  # 保留默认上方大字
        chain = dt[1:] if dt.startswith(",") else dt
        fc = bg_base_chain("0:v") + ";" + cover_chain + ";" + overlays + ";" + f"[base]{chain}[out]"
        cmd = (f'ffmpeg -y {input_src} -filter_complex "{fc}" '
               f'-map "[out]" -t {dur} -r {FPS} -pix_fmt yuv420p -movflags +faststart "{out}"')
        print("render bg (method+cover):", sc, dur)
        subprocess.run(cmd, shell=True, check=True, stderr=subprocess.DEVNULL)
        concat.append(f"file '{TAG}_bg_{sc}.mp4'")
        continue

    # ---- 普通段（hook/pain/trust 及无书封时）：原逻辑不变 ----
    input_src = bg_input(img)
    if not img:
        vf = "format=yuv420p"
    else:
        vf = ("format=yuv420p,"
              f"scale={W}:{H}:force_original_aspect_ratio=increase:flags=lanczos,crop={W}:{H},"
              f"eq=brightness=0.15:contrast=1.06,"
              f"drawbox=y=0:color=black@0.10:w={W}:h={H}:t=fill")
    vf += build_drawtext(lines)
    cmd = f'ffmpeg -y {input_src} -vf "{vf}" -t {dur} -r {FPS} -pix_fmt yuv420p -movflags +faststart "{out}"'
    print("render bg:", sc, dur)
    subprocess.run(cmd, shell=True, check=True, stderr=subprocess.DEVNULL)
    concat.append(f"file '{TAG}_bg_{sc}.mp4'")

open(os.path.join(WORK, f"{TAG}_concat_list.txt"), "w").write("\n".join(concat))

subprocess.run(
    f'ffmpeg -y -f concat -safe 0 -i "{TAG}_concat_list.txt" -c copy "{TAG}_bg_all.mp4"',
    shell=True, check=True, stderr=subprocess.DEVNULL, cwd=WORK)

# ---------- 5) 合成 ----------
final = os.path.join(WORK, OUT_NAME)
subprocess.run(
    f'ffmpeg -y -i "{TAG}_bg_all.mp4" -i "{VOICE_PATH}" '
    f'-vf "subtitles={TAG}_scene.ass,tpad=stop_mode=clone:stop_duration=10" '
    f'-c:v libx264 -b:v 1000k -maxrate 1500k -bufsize 3000k -preset medium -c:a copy -shortest -pix_fmt yuv420p -movflags +faststart "{OUT_NAME}"',
    shell=True, check=True, stderr=subprocess.DEVNULL, cwd=WORK)

dur = subprocess.check_output(
    f'ffprobe -v error -show_entries format=duration -of csv=p=0 "{OUT_NAME}"', cwd=WORK).strip()
print("FINAL:", OUT_NAME, "duration=", dur.decode())
```

## 五、灵枢策划包结构（ling_planner_pack）

`source/ling_planner_pack_<代号>_v1.json` 是灵枢产出，供灵映/渲染参考，顶层含：
`title / style / audience / variant_spec / core_script / core_script_b / platform_variants{6平台}.{a,b}.{hook,cta,hook_bg,cta_bg} / materials / notes`。
其中 `platform_variants` 是 6 平台 × A/B 的 hook/cta/背景差异来源；渲染器实际读取的是拆出来的 `topic_*.json`（由 `gen_<代号>_topics.py` 从策划包生成）。

> 换机器时，新书走完整 SOP（灵阅→灵枢→灵映）会自然产出这个策划包；若只想复刻已有书，直接拿 `topic_*.json` + `render_topic.py` 即可，无需策划包。
